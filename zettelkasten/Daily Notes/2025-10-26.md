// Weighted combination of multiple heuristics
pub struct WeightedCombinationHeuristic {
    heuristics: Vec<(Box<dyn Heuristic>, f64)>, // (heuristic, weight)
}

impl WeightedCombinationHeuristic {
    pub fn new() -> Self {
        Self { heuristics: Vec::new() }
    }
    
    pub fn add_heuristic<H: Heuristic + 'static>(&mut self, heuristic: H, weight: f64) {
        self.heuristics.push((Box::new(heuristic), weight));
    }
}

impl Heuristic for WeightedCombinationHeuristic {
    fn estimate(&self, from: NodeId, to: NodeId, context: &HeuristicContext) -> f64 {
        self.heuristics
            .iter()
            .map(|(heuristic, weight)| weight * heuristic.estimate(from, to, context))
            .sum()
    }
    
    fn name(&self) -> &str { "WeightedCombination" }
}

// Custom heuristic for terrain-aware pathfinding
pub struct TerrainAwareHeuristic {
    terrain_costs: HashMap<NodeId, f64>,
    base_heuristic: ManhattanHeuristic,
}

impl Heuristic for TerrainAwareHeuristic {
    fn estimate(&self, from: NodeId, to: NodeId, context: &HeuristicContext) -> f64 {
        let base_estimate = self.base_heuristic.estimate(from, to, context);
        let terrain_penalty = self.terrain_costs.get(&to).copied().unwrap_or(1.0);
        base_estimate * terrain_penalty
    }
}

    epsilon: f64, // Approximation factor for efficiency

impl MultiObjectiveAstar {
    pub fn find_pareto_optimal_paths(
        &self,
        graph: &dyn WeightedGraph,
        start: NodeId,
        goal: NodeId,
        context: &HeuristicContext
    ) -> Result<MultiObjectiveResult> {
        let mut pareto_frontier = Vec::new();
        let mut solutions = Vec::new();
        
        // Search for multiple non-dominated solutions
        for weight_combination in self.generate_weight_vectors() {
            let weighted_heuristic = self.create_weighted_heuristic(&weight_combination);
            let result = self.single_objective_search(graph, start, goal, weighted_heuristic)?;
            
            if self.is_non_dominated(&result, &solutions) {
                solutions.push(result);
            }
        }
        
        Ok(MultiObjectiveResult {
            paths: solutions,
            objectives: self.objective_names(),
            pareto_frontier: self.compute_pareto_values(&solutions),
        })
    }
}

    fn is_valid_edge(&self, from: NodeId, to: NodeId, context: &PathContext) -> bool;
}

// Example constraints
pub struct ForbiddenZoneConstraint {
    forbidden_nodes: HashSet<NodeId>,
}

impl PathConstraint for ForbiddenZoneConstraint {
    fn is_valid_node(&self, node: NodeId, _context: &PathContext) -> bool {
        !self.forbidden_nodes.contains(&node)
    }
    
    fn is_valid_edge(&self, _from: NodeId, to: NodeId, _context: &PathContext) -> bool {
        !self.forbidden_nodes.contains(&to)
    }
}

pub struct MaxCostConstraint {
    max_total_cost: f64,
}

impl PathConstraint for MaxCostConstraint {
    fn is_valid_node(&self, _node: NodeId, context: &PathContext) -> bool {
        context.current_cost <= self.max_total_cost
    }
    
    fn is_valid_edge(&self, _from: NodeId, _to: NodeId, context: &PathContext) -> bool {
        context.current_cost <= self.max_total_cost

use std::panic::{catch_unwind, AssertUnwindSafe};

// Robust pathfinding with panic recovery
pub fn robust_pathfinding(
    graph: &dyn WeightedGraph,
    start: NodeId,
    goal: NodeId,
    strategies: Vec<Box<dyn PathfindingStrategy>>
) -> Result<PathResult> {
    for (i, strategy) in strategies.iter().enumerate() {
        let result = catch_unwind(AssertUnwindSafe(|| {
            strategy.find_path(graph, start, goal)
        }));
        
        match result {
            Ok(Ok(path_result)) => return Ok(path_result),
            Ok(Err(pathfinding_error)) => {
                log::warn!("Strategy {} failed: {}", i, pathfinding_error);
                continue; // Try next strategy
            }
            Err(panic_payload) => {
                log::error!("Strategy {} panicked: {:?}", i, panic_payload);
                continue; // Try next strategy
            }
        }
    }
    
    Err(PathfindingError::AllStrategiesFailed)
}

// Custom panic hook for pathfinding debugging
pub fn setup_pathfinding_panic_hook() {
    std::panic::set_hook(Box::new(|panic_info| {
        log::error!("Pathfinding panic occurred!");
        
        if let Some(location) = panic_info.location() {
            log::error!("Panic at {}:{}:{}", 
                location.file(), location.line(), location.column());
        }
        
        if let Some(message) = panic_info.payload().downcast_ref::<&str>() {
            log::error!("Panic message: {}", message);
        }
        
        // Log current pathfinding state for debugging

    }));
}

use std::env;

// Configuration from environment variables
#[derive(Debug)]
pub struct PathfindingConfig {
    pub algorithm: String,
    pub heuristic: String,
    pub max_iterations: usize,
    pub timeout_ms: u64,
    pub enable_bidirectional: bool,
    pub debug_mode: bool,
}

impl Default for PathfindingConfig {
    fn default() -> Self {
        Self {
            algorithm: "astar".to_string(),
            heuristic: "manhattan".to_string(),
            max_iterations: 100_000,
            timeout_ms: 5000,
            enable_bidirectional: false,
            debug_mode: false,
        }
    }
}

impl PathfindingConfig {
    pub fn from_env() -> Self {
        let mut config = Self::default();
        
        if let Ok(algorithm) = env::var("PATHFINDING_ALGORITHM") {
            config.algorithm = algorithm;
        }
        
        if let Ok(heuristic) = env::var("PATHFINDING_HEURISTIC") {
            config.heuristic = heuristic;
        }
        
        if let Ok(max_iter) = env::var("PATHFINDING_MAX_ITERATIONS") {
            config.max_iterations = max_iter.parse().unwrap_or(config.max_iterations);
        }
        
        if let Ok(timeout) = env::var("PATHFINDING_TIMEOUT_MS") {
            config.timeout_ms = timeout.parse().unwrap_or(config.timeout_ms);
        }
        
        config.enable_bidirectional = env::var("PATHFINDING_BIDIRECTIONAL")
            .map(|v| v.to_lowercase() == "true")
            .unwrap_or(false);
            
        config.debug_mode = env::var("PATHFINDING_DEBUG")
            .map(|v| v.to_lowercase() == "true")
            .unwrap_or(false);
        
        config
    }
    
    pub fn create_pathfinder(&self) -> Box<dyn Pathfinder> {
        match self.algorithm.as_str() {
            "dijkstra" => Box::new(DijkstraPathfinder::new()),
            "astar" => {
                let heuristic = self.create_heuristic();
                if self.enable_bidirectional {
                    Box::new(BidirectionalAstar::new(heuristic))
                } else {
                    Box::new(AstarPathfinder::new(heuristic))
                }
            }
            _ => Box::new(AstarPathfinder::new(ManhattanHeuristic)),
        }
    }
}
