// Use more cache-friendly data structures
use std::collections::VecDeque;

struct OptimizedPathfinding {
    // Use Vec instead of HashMap where possible (better cache locality)
    distances: Vec<Distance>,  // Index by NodeId
    visited: Vec<bool>,         // Bitset for visited nodes

    // Pool memory to reduce allocations
    node_pool: Vec<Node>,

    // Reuse priority queue between searches
    pq: PriorityQueue<(Distance, NodeId)>,
}

impl OptimizedPathfinding {
    fn clear_for_reuse(&mut self) {
        // Reset state for next search without reallocating
        self.distances.fill(Distance::MAX);
        self.visited.fill(false);
        self.pq.clear();
    }
}

use anyhow::{Result, Context};

fn optimized_pathfinding(
    graph: &Graph,
    start: NodeId,
    goal: NodeId
) -> Result<Vec<NodeId>>
{
    // Chain operations with combinators
    validate_nodes(graph, start, goal)
        .context("Invalid start or goal node")?;

    bidirectional_search(graph, start, goal)
        .or_else(|_| fallback_astar(graph, start, goal))
        .and_then(|path| {
            verify_path(graph, &path)
                .map(|_| path)
        })
        .map_err(|e| {
            log_pathfinding_error(&e);
            e
        })
}

// Using combinators to transform errors
fn try_multiple_strategies(graph: &Graph, start: NodeId, goal: NodeId)
