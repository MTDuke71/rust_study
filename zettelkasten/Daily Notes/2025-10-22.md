// Mission 9 architecture design patterns
mod pathfinding {
    // Core pathfinding traits
    pub trait Pathfinder<G: WeightedGraph> {
        type Error;
        
        fn find_path(&self, graph: &G, start: G::NodeId, goal: G::NodeId) 
            -> Result<Vec<G::NodeId>, Self::Error>;
        
        fn find_shortest_path(&self, graph: &G, start: G::NodeId, goal: G::NodeId) 
            -> Result<(Vec<G::NodeId>, f64), Self::Error>;
    }
    
    // Heuristic function trait for A*
    pub trait Heuristic<G: WeightedGraph> {
        fn estimate(&self, from: G::NodeId, to: G::NodeId, graph: &G) -> f64;
    }
    
    // Weighted graph trait for pathfinding algorithms
    pub trait WeightedGraph {
        type NodeId: Copy + Eq + Hash;
        type Weight: Copy + Add<Output = Self::Weight> + PartialOrd;
        
        fn neighbors(&self, node: Self::NodeId) -> Vec<(Self::NodeId, Self::Weight)>;
        fn node_count(&self) -> usize;
    }
}

// Custom error types for pathfinding
#[derive(Debug, thiserror::Error)]
pub enum PathfindingError {
    #[error("No path exists from {start:?} to {goal:?}")]
    NoPathExists { start: NodeId, goal: NodeId },
    
    #[error("Invalid node: {node:?}")]
    InvalidNode { node: NodeId },
    
    #[error("Graph is empty or malformed")]
    InvalidGraph,
    
    #[error("Heuristic calculation failed: {reason}")]
    HeuristicError { reason: String },
    
    #[error("Priority queue error: {0}")]
    QueueError(String),
}

// CLI argument parsing for pathfinding tools
use std::env;

fn parse_pathfinding_args() -> Result<PathfindingConfig, Box<dyn std::error::Error>> {
    let args: Vec<String> = env::args().collect();
    
    if args.len() < 4 {
        return Err("Usage: pathfinder <algorithm> <start> <goal> [graph_file]".into());
    }
    
    let algorithm = match args[1].as_str() {
        "dijkstra" => Algorithm::Dijkstra,
        "astar" => Algorithm::AStar,
        _ => return Err("Algorithm must be 'dijkstra' or 'astar'".into()),
    };
    
    let start: usize = args[2].parse()?;
    let goal: usize = args[3].parse()?;
    let graph_file = args.get(4).map(String::clone);
    
    Ok(PathfindingConfig { algorithm, start, goal, graph_file })
}

struct PathfindingConfig {
    algorithm: Algorithm,
    start: usize,
    goal: usize,
    graph_file: Option<String>,
}

enum Algorithm {
    Dijkstra,
    AStar,
}

// Custom error type practice
#[derive(Debug)]
struct PathfindingError {
    kind: ErrorKind,
    message: String,
    source: Option<Box<dyn std::error::Error>>,
}

impl std::fmt::Display for PathfindingError {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        write!(f, "Pathfinding error ({}): {}", self.kind, self.message)
    }
}

impl std::error::Error for PathfindingError {
    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
        self.source.as_ref().map(|e| e.as_ref())
    }
}

#[derive(Debug)]
enum ErrorKind {
    NoPath,
    InvalidInput,
    GraphError,
    HeuristicError,
}
